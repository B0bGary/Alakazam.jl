function draw_ind_graph(g1)
    pos = nx.shell_layout(g1)  # Layout algorithm, you can choose other layouts too
    nx.draw(g1, pos, with_labels=false, node_color="skyblue", node_size=800, font_size=10, font_color="black")
    # Extract edge weights as labels
    edge_labels = Dict{Tuple{Int,Int},Float64}()
    for (u, v, d) in g1.edges(data=true)
        edge_labels[(u, v)] = d["weight"]
    end
    println(g1.nodes(data=true))
    for n in g1.nodes(data=true)
        println(n)
        println(n[2])
        println(n[2]["name"])
    end
    node_labels = Dict(n[1] => string(n[1]) * ":" * string(n[2]["name"]) for n in g1.nodes(data=true))

    # Draw node labels (custom or default)
    nx.draw_networkx_labels(g1, pos, labels=node_labels, font_size=12, font_color="black")

    # Add edge weight labels to the graph
    nx.draw_networkx_edge_labels(g1, pos, edge_labels=edge_labels, font_size=10)
    plt.title("Tensor")
    plt.show()
end

function factor_2_3(n::Int)
    n < 1 && error("n must be positive")

    n2 = 0
    n3 = 0

    while iseven(n)
        n >>>= 1
        n2 += 1
    end

    while n % 3 == 0
        n ÷= 3
        n3 += 1
    end

    n == 1 || error("has primes other than 2 and 3")

    return n2, n3
end

# Function to find cycles of a given length
function find_cycles_of_length(g, cycle_length)
    cycles = []

    function dfs(node, start, visited, stack)
        push!(stack, node)
        visited[node] = true

        if length(stack) == cycle_length
            if pycall(g.has_edge, Bool, stack[end], start)
                push!(cycles, copy(stack))
            end
        else
            for neighbor in pycall(g.neighbors, PyObject, node)
                if !haskey(visited, neighbor) || !visited[neighbor]
                    dfs(neighbor, start, visited, stack)
                end
            end
        end

        pop!(stack)
        visited[node] = false
    end

    nodes = pycall(g.nodes, PyObject) |> collect

    for node in nodes
        visited = Dict(n => false for n in nodes)  # Initialize visited dict for each start node
        dfs(node, node, visited, [])
    end

    return cycles
end

function find_weighted_square(g)
    target_weights = [-1, 2, 1, 2]
    four_cycles = find_cycles_of_length(g, 4)

    for cycle in four_cycles
        # println(cycle)
        edges = [(cycle[i], cycle[mod1(i + 1, 4)]) for i in 1:4]

        # Extract weights
        weights = []
        for (u, v) in edges
            edge_data = pycall(g.get_edge_data, PyObject, u, v)  # Fetch edge data
            # println("Edge data: ", edge_data)
            # println("Edge ($u, $v): ", edge_data)  # Debug output

            if edge_data !== nothing
                edge_data_dict = PyDict(edge_data) # Convert edge data to Dict
                # println("Edge data dict: ", edge_data_dict)
                found_weight = false
                for (key, edata) in pairs(edge_data_dict)  # Iterate through all edges between u and v
                    # println("edata: ", edata)
                    if isa(edata, Dict) && haskey(edata, "weight")
                        push!(weights, edata["weight"])
                        found_weight = true
                        break  # Use the first available weight
                    end
                end
                if !found_weight
                    push!(weights, missing)  # No weight found
                end
            else
                push!(weights, missing)  # No edge data found
            end
        end
        # println(weights)

        # **Fix: Ensure no missing values before checking weights**
        if all(x -> !ismissing(x), weights) && any(isequal(weights, circshift(target_weights, i)) for i in 0:3)
            # println("Found matching square in nodes: ", cycle)
            return true#cycle
        else
            nothing
            # println("No cycle in this square")
        end
    end

    # println("No matching square found")
    return false
end



# Function for cyclic shift
function circshift(arr, n)
    n = n % length(arr)
    return vcat(arr[n+1:end], arr[1:n])
end

# Function to find edges with a specific weight
function edges_with_weight(g, vertex_id, target_weight)
    println("checking edges with weight ", target_weight, " for vertex ", vertex_id)
    edges = pycall(g.edges, PyObject, vertex_id, data="weight")
    linked_vertex_ids = []

    for edge in edges
        u, v, weight = edge
        if weight == target_weight
            if u != vertex_id
                push!(linked_vertex_ids, u)
            end
            if v != vertex_id
                push!(linked_vertex_ids, v)
            end
        end
    end

    return linked_vertex_ids
end


## finds the overall sign change when permuting graded tensors with a particular perm
function perm_grading(perm_dict::Dict{Int,Int}, tt::TensorTerm)
    n = length(tt.terms)

    # Initialize inversion count
    inv_count = 0

    # Loop over each pair of indices in the original permutation
    for i in 1:n-1
        for j in i+1:n
            # Check if the elements are in reverse order in the permutation (i.e., an inversion)
            # println(perm_dict)
            if perm_dict[i] > perm_dict[j]
                # Check if the grading of elements at i and j are different
                if grading(tt.terms[i]) * grading(tt.terms[j]) == 1
                    inv_count += 1
                end
            end
        end
    end

    # If the number of inversions is even, return true (even permutation), else false (odd permutation)
    return inv_count % 2 # to then do (-1)^perm_grading

end

    end

    dums1 = get_dummies(T1)
    dums2 = get_dummies(T2)
    #if we dont need to worry about dummies, do simple algo

function do_swaps(T1::TensorSuperType, patt::OrderedSet, movers::Dict)

    # return 0 ##TEMP disabled tableau since implements wrong for general tabs
    #where {I<:IndexSuperType}
    #elemetns in patt: indiceobj, intpos
    #elemtns in mover: intpos, indiceobj
    if length(patt) == 1 || length(patt) == 0
        return 1 #no swap needed
    else
        #loop over movers, see if a move exists
        for (intpos, indiceobj) in movers
            if patt[1][1] == indiceobj #same index, different pos
                for tab in T1.tableaux
                    # println(patt[1][2])
                    if tab.adj_matrix[patt[1][2], intpos] != 0
                        grad = tab.adj_matrix[patt[1][2], intpos]
                        movers[intpos] = movers[patt[1][2]]
                        delete!(movers, patt[1][2])
                        popfirst!(patt)
                        return grad * do_swaps(T1, patt, movers)
                        break
                    end

                end
            else
                continue
            end
        end
        #no matches
        return 0
    end
end

function sign_of_permutation(perm, perm_matrix)
    sign = 1
    for i in 1:length(perm)
        for j in (i+1):length(perm)
            if perm[i] > perm[j]
                sign *= perm_matrix[indices[i], indices[j]]
            end
        end
    end
    return sign
end
